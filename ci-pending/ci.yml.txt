name: CI

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-dotnet@v4
        with:
          dotnet-version: '8.0.x'

      - name: Build
        run: dotnet build src/SRPG.Battle.Core/SRPG.Battle.Core.csproj --configuration Release

      - name: Test
        run: dotnet test tests/SRPG.Battle.Core.Tests/SRPG.Battle.Core.Tests.csproj --configuration Release --verbosity normal
